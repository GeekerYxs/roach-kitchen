def stamp():return datetime.now(ZoneInfo('Asia/Shanghai')).isoformat(timespec='seconds')
def log(f,event,**kw):f.write(json.dumps({'time':stamp(),'event':event,**kw},ensure_ascii=False)+'\n');f.flush()
def catchable_names():
 with MONSTERS.open(encoding='utf-8-sig',newline='') as f:return {r['name'] for r in csv.DictReader(f) if r.get('catchable')=='1'}
def tap_seen(offset,target,slot):
 if not TAP.exists():return False
 with TAP.open('rb') as f:
  f.seek(offset)
  for raw in f:
   try:e=json.loads(raw.decode('utf-8'))
   except Exception:continue
   if e.get('kind')=='request' and e.get('opcode')==OP_ITEM_TO_FIGHTER:
    try:b=bytes.fromhex(e.get('hex',''))
    except ValueError:continue
    if len(b)>=8 and b[5]==target and b[7]==slot:return True
 return False
